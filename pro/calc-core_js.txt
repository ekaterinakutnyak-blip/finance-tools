/* ============================================================
   calc-core.js — единый источник правды для всех расчётов по облигациям.
   Никакого DOM. Только чистые функции. Дёргается и скринером, и
   (в перспективе) остальными калькуляторами — чтобы формула налога,
   спреда и агрегации жила в ОДНОМ месте и не разъезжалась.
   ============================================================ */
(function (root) {
  'use strict';

  // ---------- Рейтинг: лесенка и «худший из агентств» ----------
  const LADDER = ['D','C','CC','CCC-','CCC','CCC+','B-','B','B+','BB-','BB','BB+',
                  'BBB-','BBB','BBB+','A-','A','A+','AA-','AA','AA+','AAA'];
  const RANK = {}; LADDER.forEach((g,i)=>RANK[g]=i);   // выше = надёжнее

  function normalizeRating(raw) {                       // мультиагентная строка -> худший
    if (!raw || typeof raw !== 'string') return { grade:null, rank:null };
    const ranks = [];
    raw.split(/[;\n,/]+/).forEach(tok => {
      let s = tok.trim().replace(/\(RU\)/ig,'').replace(/\.ru$/i,'').replace(/^ru/i,'').trim().toUpperCase();
      const m = s.match(/^([ABCD]{1,3}[+-]?)$/);
      if (m && (m[1] in RANK)) ranks.push(RANK[m[1]]);
    });
    if (!ranks.length) return { grade:null, rank:null };
    const r = Math.min(...ranks);
    return { grade: LADDER[r], rank: r };
  }

  // ---------- Налог на купон: ступенька 13% / 15% ----------
  // По НДФЛ 2025+: до 2,4 млн налогооблагаемого дохода в год — 13%, свыше — 15%.
  function taxOnCoupon(grossAnnual, opts) {
    const limit = (opts && opts.bracket) || 2_400_000;
    const lo = Math.min(grossAnnual, limit) * 0.13;
    const hi = Math.max(0, grossAnnual - limit) * 0.15;
    return lo + hi;                                     // сумма налога, ₽
  }
  function netCoupon(grossAnnual, opts) { return grossAnnual - taxOnCoupon(grossAnnual, opts); }

  // ---------- YTM 30/360 (для живого слоя MOEX) ----------
  function days360(d1, d2) {                            // US NASD
    let dd1 = Math.min(d1.getDate(),30), dd2 = d2.getDate();
    if (dd2===31 && dd1===30) dd2=30;
    return (d2.getFullYear()-d1.getFullYear())*360 + (d2.getMonth()-d1.getMonth())*30 + (dd2-dd1);
  }
  function yearfrac(d1,d2){ return days360(d1,d2)/360; }
  function ytm30360(dirtyRub, cashflows /*[{date:Date,amount}]*/, settle) {
    const fut = cashflows.filter(c => c.date > settle);
    if (!fut.length || dirtyRub<=0) return null;
    const npv = r => fut.reduce((s,c)=> s + c.amount/Math.pow(1+r, yearfrac(settle,c.date)), 0) - dirtyRub;
    let lo=-0.9, hi=5.0, flo=npv(lo), fhi=npv(hi);
    if (flo*fhi>0) return null;
    for (let i=0;i<200;i++){ const m=(lo+hi)/2, fm=npv(m);
      if (Math.abs(fm)<1e-6) return m;
      if (flo*fm<0){hi=m;} else {lo=m;flo=fm;} }
    return (lo+hi)/2;
  }
  function curveYield(curve /*{years:yield%}*/, years) {
    const xs = Object.keys(curve).map(Number).sort((a,b)=>a-b);
    if (years<=xs[0]) return curve[xs[0]];
    if (years>=xs[xs.length-1]) return curve[xs[xs.length-1]];
    for (let i=1;i<xs.length;i++) if (years<=xs[i]) {
      const x0=xs[i-1],x1=xs[i],y0=curve[x0],y1=curve[x1];
      return y0+(y1-y0)*(years-x0)/(x1-x0);
    }
    return curve[xs[xs.length-1]];
  }
  function gSpreadBp(ytmPct, durationYears, curve) {    // спред в базисных пунктах
    return Math.round((ytmPct - curveYield(curve, durationYears))*100*10)/10;
  }

  // ---------- Флаг аномального спреда (риск-first) ----------
  // спред подозрительно велик для своего рейтинга => рынок пахнет дефолтом
  function markAnomalous(bonds, k) {
    k = k || 1.75;
    const byRank = {};
    bonds.forEach(b => { if (b.grank!=null && b.gspread!=null) (byRank[b.grank]=byRank[b.grank]||[]).push(b.gspread); });
    const med = {};
    Object.keys(byRank).forEach(r => { const a=byRank[r].slice().sort((x,y)=>x-y); med[r]=a[Math.floor(a.length/2)]; });
    bonds.forEach(b => {
      b.anom = false;
      if (b.grank!=null && b.gspread!=null && med[b.grank]>0)
        b.anom = b.gspread > Math.max(med[b.grank]*k, med[b.grank]+3); // спред в %, +3 пп абс.порог
    });
    return bonds;
  }

  // ---------- Фильтр вселенной ----------
  // p: { currencies:[...], allowFloater, allowLinker, allowStep, allowAmort, allowPutCall,
  //      allowSubord, allowQual, minListing, spreadMin, spreadMax, ratingFloorRank,
  //      allowNoRating, excludeAnom, excludeWarn, termMin, termMax, durMin, durMax, minLiq }
  function filterUniverse(bonds, p) {
    p = p || {};
    return bonds.filter(b => {
      // структура
      if (p.currencies && p.currencies.length && !p.currencies.includes(b.cur)) return false;
      if (!p.allowFloater && b.fFloat) return false;
      if (!p.allowLinker  && b.fLinker) return false;
      if (!p.allowStep    && b.fStep) return false;
      if (!p.allowAmort   && b.fAmort) return false;
      if (!p.allowPutCall && (b.fPut || b.fCall)) return false;
      if (!p.allowSubord  && b.subord) return false;
      if (!p.allowQual    && b.qual) return false;
      if (p.minListing && (b.listing==null || +b.listing > +p.minListing)) return false; // 1 лучше 3
      if (p.excludeWarn && b.fWarn) return false;
      // ликвидность
      if (p.minLiq!=null && (b.liq==null || b.liq < p.minLiq)) return false;
      // рейтинг
      if (b.grank==null) { if (!p.allowNoRating) return false; }
      else if (p.ratingFloorRank!=null && b.grank < p.ratingFloorRank) return false;
      // риск: коридор G-спреда — ТОЛЬКО для рублёвых бумаг.
      // У валютных G-спреда к рублёвой кривой ОФЗ нет, этот фильтр их не касается
      // (их риск — валютный, показан отдельно). Иначе они молча выпадали из отбора.
      if (b.cur === 'Рубль') {
        if (p.spreadMin!=null && (b.gspread==null || b.gspread < p.spreadMin)) return false;
        if (p.spreadMax!=null && (b.gspread==null || b.gspread > p.spreadMax)) return false;
        if (p.excludeAnom && b.anom) return false;
      }
      // срок
      if (p.termMin!=null && (b.term==null || b.term < p.termMin)) return false;
      if (p.termMax!=null && (b.term==null || b.term > p.termMax)) return false;
      if (p.durMin!=null && (b.dur==null || b.dur < p.durMin)) return false;
      if (p.durMax!=null && (b.dur==null || b.dur > p.durMax)) return false;
      return true;
    });
  }

  // ---------- Агрегация портфеля (равные веса выживших) ----------
  // Купон агрегируем ПО ВАЛЮТАМ (YTM наверх НЕ поднимаем — он не складывается).
  function aggregate(selected, capital, opts) {
    opts = opts || {};
    const n = selected.length;
    if (!n || capital<=0) return null;
    const w = 1/n, amt = capital/n;
    const byCur = {};        // валюта -> {grossCoupon, weight}
    let wSpread=0, wDur=0, worst=null, spN=0, durN=0, anom=0;
    const sectors={}, issuers={};
    selected.forEach(b => {
      const cur = b.cur || 'Рубль';
      byCur[cur] = byCur[cur] || { grossCoupon:0, weight:0, n:0 };
      if (b.tkd!=null) byCur[cur].grossCoupon += amt * b.tkd/100;   // купон в валюте бумаги
      byCur[cur].weight += w; byCur[cur].n++;
      if (b.gspread!=null){ wSpread += b.gspread*w; spN++; }
      if (b.dur!=null){ wDur += b.dur*w; durN++; }
      if (b.grank!=null) worst = (worst==null)? b.grank : Math.min(worst,b.grank);
      if (b.anom) anom++;
      const s=b.sector||'н/д', i=b.issuer||b.name; sectors[s]=(sectors[s]||0)+w; issuers[i]=(issuers[i]||0)+w;
    });
    // рублёвый доход чистыми (с налоговой ступенькой); валютные — в их валюте, отдельно
    const rub = byCur['Рубль'] ? byCur['Рубль'].grossCoupon : 0;
    const income = {};
    Object.keys(byCur).forEach(cur => {
      const gross = byCur[cur].grossCoupon;
      const net = (cur==='Рубль') ? netCoupon(gross, opts) : gross; // налог считаем для ₽; валюту — брутто (курс отдельно)
      income[cur] = { grossYear:gross, netYear:net, netMonth:net/12, share: byCur[cur].weight };
    });
    const maxSector = Math.max(...Object.values(sectors));
    const maxIssuer = Math.max(...Object.values(issuers));
    return {
      n, income,
      wavgSpread: spN? +wSpread.toFixed(0):null,
      wavgDur: durN? +wDur.toFixed(2):null,
      worstGrade: worst!=null? LADDER[worst]:null,
      anomCount: anom,
      maxSector: +(maxSector*100).toFixed(0), maxIssuer:+(maxIssuer*100).toFixed(0),
      sectors, issuers,
      flags: buildFlags(anom, maxSector, maxIssuer)
    };
  }
  function buildFlags(anom, maxSector, maxIssuer) {
    const f=[];
    if (anom>0) f.push({t:'danger', m:`аномальный спред у ${anom} бумаг — рынок закладывает риск выше рейтинга`});
    if (maxSector>0.35) f.push({t:'warn', m:`${Math.round(maxSector*100)}% в одной отрасли — концентрация`});
    if (maxIssuer>0.15) f.push({t:'warn', m:`${Math.round(maxIssuer*100)}% в одном эмитенте`});
    return f;
  }

  root.CalcCore = {
    LADDER, RANK, normalizeRating,
    taxOnCoupon, netCoupon,
    days360, yearfrac, ytm30360, curveYield, gSpreadBp,
    markAnomalous, filterUniverse, aggregate
  };
})(typeof window!=='undefined' ? window : globalThis);
